## 3. How Workflows Work in the TS Code (Step 2)

### 3.1 What is a "Workflow"?

A workflow is a DAG of tasks defined by Zod-validated payloads, not a class hierarchy:

```12:34:packages/types/src/domain/workflows/WorkflowPayload.ts
export const WorkflowPayloadSchema = z
    .strictObject({
        name: z.string().min(1).max(256),
        dag: z.array(TaskInWorkflowInstanceSchema),
        artifact_groups: z.array(ArtifactGroupSchema).nullable().optional(),
        uid: z.string().nullable().optional(),
        path_prefix: z.string().nullable().optional(),
    })
```

Orchestration roles:

- `OperationWorkflowAdapter`—maps domain `Operation` → `WorkflowPayload` (`BaseOperationWorkflowAdapter.ts` lines 17–40)
- `WorkflowManager`—submits + polls (`WorkflowManager.ts` lines 31–48, 50–185)
- `DistributedWorkflowInstance`—persisted multi-node DAG with node state (`DistributedWorkflowInstance.ts` lines 27–120)
- `DistributedWorkflowSupervisor`—global submit/poll scheduler (`DistributedWorkflowSupervisor.ts` lines 56–72)

### 3.2 Sequencing & Triggering

| Layer | Mechanism |
|-------|-----------|
| Single operations | Direct HTTP invocation → `OperationWorkflowManager.run()` → adapter → `WorkflowManager.startWorkflow()` |
| Status tracking | Cron-scheduled polling (`WorkflowManager.startPolling`, lines 124–185) |
| Distributed workflows | Supervisor `tick()` loop: priority submit heap + poll heap with exponential backoff (lines 130–317) |
| Optout responses | HTTP trigger → background `OptoutResponseProcessorService` → creates new operations |
| Argo execution | workflows-api builds Hera DAG and calls Argo API (`workflow_builder.py` lines 24–48) |

### 3.3 State Persistence

| State | Store |
|-------|-------|
| `OperationTracking` | MongoDB (`OperationTrackingRepo.ts`)—status, receipt, correlation ID |
| `DistributedWorkflowInstance` | MongoDB—per-node status, tracking IDs, DAG |
| `WorkflowTracking` / `DatasetTracking` | MongoDB—post-finalize artifacts |
| Argo runtime | Argo Workflows API (external)—pod execution state |
| In-flight supervisor queues | In-memory heaps (`DistributedWorkflowSupervisor.ts` lines 61–72) |

### 3.4 Errors, Retries, Failure Handling

- WorkflowManager polling: transient network errors retry; terminal errors synthesize `Failed` response (`WorkflowManager.ts` lines 137–176)
- Supervisor submission: tenant gateway network errors leave node `Pending` for retry; other errors call `failNode()` (lines 177–190)
- Supervisor polling: exponential backoff + circuit breaker on repeated failures (lines 288–309, `CircuitBreaker.ts`)
- Operation level: `handleWorkflowResponse` calls `adapter.onWorkflowFailure()` then `operationTracking.fail()` (lines 298–304)
- Argo submit: Python retry wrapper for 5xx (`workflow_submitting.py` lines 8–26)
- Legacy Python: duplicate function definition for `submit_load_ude_and_uniquify_by_primary_key` (lines 188–200)—second definition silently overrides first

### 3.5 End-to-end Lifecycle: Optout Response Workflow

```
POST /opt-out/process-responses
  → OptoutTrackingController.triggerProcessResponses (routes/optout/)
  → OptoutResponseProcessorService.processResponsesFromOptoutSource
      → Mesh poll, map message IDs to prior OptoutRequest operations
      → storeOptoutResponse (S3)
      → OperationWorkflowManager.run(HandleOptoutResponse operation)
          → HandleOptoutResponseOperationWorkflowAdapter.createWorkflowPayload (lines 41–64)
          → WorkflowManager.startWorkflow → workflows-api
              → opt_out_mesh_response Argo template (workflow_submitting.py:125–129)
          → OperationTracking saved with receipt
  → WorkflowManager cron poll until Succeeded/Failed
  → OperationWorkflowManager.handleWorkflowResponse (lines 264–328)
      → FinalizeOptoutWorkflowUseCase.execute (FinalizeOptoutWorkflowUseCase.ts:48–128)
          → compare row counts, supersede dataset, S3 delete, save tracking
```

#### File 3 (New): `apps/fitconnect/src/services/workflows/FinalizeDatasetMetadataService.ts`

Extract S3-dependent logic from `FinalizeService.fetchDatasetMetadata` and `getOutputDatasetSchema`.

```typescript
import { DataSetSchemaVO } from '@domain/dataSource/schema/DataSetSchemaVO';
import { DatasetTracking } from '@domain/query/dataset_tracking/DatasetTracking';
import { DataSetTrackingBuilder } from '@domain/query/dataset_tracking/DatasetTrackingBuilder';
import { BucketManager, GetObjectResponseType } from '@infra/s3/BucketManager';
import { PartialDatasetSchemaSchema } from '@fitfile/types';
import Logger from 'bunyan';
import { Err, Ok, Result } from 'ts-results';
import { basename } from 'path';
import { FinalizeDataset } from './FinalizeMetadata';
import {
    DatasetFileNotFound,
    MissingFinalizeMetadata,
    SchemaFileNotFound,
} from './FinalizeService'; // re-export errors, or move to FinalizeErrors.ts

export type FinalizeDatasetMetadataServiceProps = {
    s3Client: BucketManager;
    logger: Logger;
};

export class FinalizeDatasetMetadataService {
    constructor(private readonly props: FinalizeDatasetMetadataServiceProps) {}

    async fetch(finalizeDataset: FinalizeDataset): Promise<Result<DatasetTracking, Error>> {
        const datasetId = DatasetTracking.createOutputDatasetId(finalizeDataset.datasetFile.path);

        let schema: DataSetSchemaVO;
        try {
            const rawSchema = await this.getOutputDatasetSchema(finalizeDataset.schemaFile.path);
            schema = new DataSetSchemaVO({ dataSetId: datasetId, schema: rawSchema });
        } catch (e) {
            this.props.logger.warn(
                { schema_file: finalizeDataset.schemaFile.path },
                'Failed to retrieve schema for output dataset',
            );
            return Err(new SchemaFileNotFound(finalizeDataset.schemaFile.path, { cause: e }));
        }

        const required = ['fields', 'record_count'] as const;
        const missing = required.filter((k) => !(k in finalizeDataset.datasetFile.attributes));
        if (missing.length > 0) {
            return Err(new MissingFinalizeMetadata(finalizeDataset.datasetFile.path, […missing]));
        }

        return Ok(
            DataSetTrackingBuilder.buildWorkflowOutputDataset({
                dataset_id: datasetId,
                data_source_id: datasetId.dataSourceId.toValue(),
                dataset_file: finalizeDataset.datasetFile.props,
                schema,
                fields: finalizeDataset.datasetFile.attributes.fields,
                record_count: finalizeDataset.datasetFile.attributes.record_count,
                display_name: basename(datasetId.datasetKey),
            }),
        );
    }

    private async getOutputDatasetSchema(schemaKey: string) {
        const schemaData = await this.props.s3Client.getObject(schemaKey, {
            responseType: GetObjectResponseType.DecompressedGzip,
        });
        if (!schemaData) throw new Error(`Failed to get output schema from s3 bucket: ${schemaKey}`);
        const schema = JSON.parse(schemaData);
        const parsed = PartialDatasetSchemaSchema.safeParse(schema);
        if (!parsed.success) throw new Error(`Output schema is not a schema format: ${schemaKey}`);
        return parsed.data;
    }
}
```

#### File 5 (New): `apps/fitconnect/src/services/optout/OptoutDatasetSupersedePolicy.ts`

Pure logic extracted from `FinalizeOptoutWorkflowUseCase` lines 85–102 and 169–183.

```typescript
import { DatasetTracking } from '@domain/query/dataset_tracking/DatasetTracking';
import { ArtifactBusinessName } from '@fitfile/types';
import { FinalizeMetadata } from '@services/workflows/FinalizeMetadata';

export type OptoutSupersedeDecision =
    | { action: 'no_change'; datasetKeyToDelete: string }
    | { action: 'supersede'; recordsRemovedCount: number; previousDatasetKeyToDelete: string };

export class OptoutDatasetSupersedePolicy {
    static decide(
        latest: DatasetTracking,
        previous: DatasetTracking,
    ): OptoutSupersedeDecision {
        if (latest.record_count === previous.props.record_count) {
            return { action: 'no_change', datasetKeyToDelete: latest.id.datasetKey };
        }

        const recordsRemovedCount =
            typeof previous.props.record_count === 'number'
                ? previous.props.record_count - latest.record_count
                : 0;

        return {
            action: 'supersede',
            recordsRemovedCount,
            previousDatasetKeyToDelete: previous.id.datasetKey,
        };
    }

    / Strip dataset/schema artifacts from metadata when no rows were removed. */
    static stripOutputArtifacts(finalizeMetadata: FinalizeMetadata): void {
        finalizeMetadata.remove([ArtifactBusinessName.DATASET, ArtifactBusinessName.SCHEMA]);
    }
}
```
